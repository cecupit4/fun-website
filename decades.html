albums = FileAttachment("top_1000_albums.csv").csv({ typed: true })

// Add decade attribute to each album
albumsWithDecade = albums.map(d => ({
  ...d,
  decade: `${Math.floor(d.year / 10) * 10}s`,
  primary_genre: d.genres ? d.genres.split(",")[0].trim() : "other"
}))

// Decade selection
viewof selectedDecade = Inputs.radio(
  ["1950s", "1960s", "1970s", "1980s", "1990s", "2000s", "2010s", "2020s"],
  { label: "Select Era", value: "1970s" }
)

// Filter albums in active decade
decadeAlbums = albumsWithDecade.filter(d => d.decade === selectedDecade)

// The top highlight album of this decade
topDecadeAlbum = decadeAlbums[0] || albumsWithDecade[0]
