Cover Showdown Game

Test your musical knowledge! Guess whether the mystery album is ranked HIGHER or LOWER in the all-time top 1000 list.

albums = FileAttachment("top_1000_albums.csv").csv({ typed: true })

cleanAlbums = albums.map(d => ({
  ...d,
  primary_genre: d.genres ? d.genres.split(",")[0].trim() : "other"
}))

// Game State: current streak, best streak, round, albumA, albumB, status
mutable currentStreak = 0
mutable bestStreak = 0
mutable gameStatus = "playing" // "playing", "won", "lost"
mutable roundCounter = 1

// Random seed triggers
function getRandomAlbum(excludeRank) {
  let item = cleanAlbums[Math.floor(Math.random() * cleanAlbums.length)];
  while (excludeRank && item.rank === excludeRank) {
    item = cleanAlbums[Math.floor(Math.random() * cleanAlbums.length)];
  }
  return item;
}

mutable gameAlbumA = cleanAlbums[Math.floor(Math.random() * 200)]
mutable gameAlbumB = getRandomAlbum(gameAlbumA.rank)

</div>

<div class="game-arena text-center">

    </div>

      🏆 Best Streak: <span style="font-size: 1.25rem;">${bestStreak}</span>
    </div>
  </div>

  <div class="compare-grid" style="margin: 1.5rem 0;">
    <!-- Album A (Known) -->
    <div class="compare-box">

      <img

        MYSTERY ALBUM

        class="compare-cover"

          mutable currentStreak = newStreak;

        mutable gameAlbumB = getRandomAlbum(gameAlbumB.rank);
        mutable gameStatus = "playing";
        mutable roundCounter = roundCounter + 1;
      }}>
        ${gameStatus === 'won' ? '🔥 Keep Streak Going →' : '🔄 Play Again'}
      </button>
    </div>
  `}
</div>
