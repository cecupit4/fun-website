Top 1000 Album Covers

Explore all 1,000 greatest albums of all time loaded directly inside a grand interactive vinyl turntable.

albums = FileAttachment("top_1000_albums.csv").csv({ typed: true })

cleanData = albums.map(d => ({
  ...d,
  primary_genre: d.genres ? d.genres.split(",")[0].trim() : "other",
  decade: `${Math.floor(d.year / 10) * 10}s`
}))

// State for active album on the needle
mutable activeAlbum = cleanData[0]
mutable isSpinning = true
viewof searchQuery = Inputs.text({
  placeholder: "Search 1,000 albums on the turntable..."
})

genresList = ["All", "rock", "pop", "folk", "electronica", "punk", "hip-hop", "soul", "jazz", "country", "metal"]
viewof selectedGenre = Inputs.select(genresList, {
  label: "Platter Genre",
  value: "All"
})
q = searchQuery.toLowerCase().trim()
matchingSet = new Set(
  cleanData
    .filter(d => {
      const matchGenre = selectedGenre === "All" || d.primary_genre === selectedGenre;
      const matchText = !q || 
        d.album.toLowerCase().includes(q) || 
        d.artist.toLowerCase().includes(q) || 
        String(d.year).includes(q) ||
        d.primary_genre.toLowerCase().includes(q);
      return matchGenre && matchText;
    })
    .map(d => d.rank)
)

spotlight = activeAlbum || cleanData[0]
// The Grand Hi-Fi Turntable Component: Big Turntable with all album covers inside of it!
html`
<div class="grand-turntable-deck">
  <!-- Top Bezel: Navigation & Brand Name -->
  <div class="turntable-bezel">
    <div class="turntable-logo-text">
      AUDIO-TECHNICA // DIRECT DRIVE HI-FI 1000
    </div>

    <div class="deck-nav-bar" style="margin-bottom: 0; padding-bottom: 0; border-bottom: none;">
      <span style="font-size: 0.72rem; font-weight: 700; color: #ffd166; text-transform: uppercase;">VAULT TABS:</span>
      <a href="index.qmd" class="deck-nav-pill active">🎵 Turntable Deck</a>
      <a href="compare.qmd" class="deck-nav-pill">⚖️ Cover Compare ↗</a>
      <a href="game.qmd" class="deck-nav-pill">🎮 Higher / Lower Game ↗</a>
      <a href="about.qmd" class="deck-nav-pill">ℹ️ About ↗</a>
    </div>
  </div>

  <!-- Turntable Body: Giant Platter containing Album Covers on Left, Tonearm on Right -->
  <div class="turntable-body-grid">
    <!-- Big Circular Platter filled with the 1,000 album covers -->
    <div class="big-platter-well">
      <!-- Center Spindle Hub with spinning active album label -->
      <div class="center-spindle-hub">
        <div class="center-spinning-vinyl" style="${isSpinning ? '' : 'animation-play-state: paused;'}">
          <img 
            src="${spotlight.image_url}" 
            alt="${spotlight.album}" 
            referrerpolicy="no-referrer"
            crossorigin="anonymous"
            onerror="this.src='https://placehold.co/300x300/181b26/00f2fe?text=Vinyl'"
          />
        </div>
        <div class="spindle-pin"></div>
      </div>

      <!-- Inside the Platter: The 1,000 Album Covers Mosaic -->
      <div class="platter-covers-mosaic">
        ${cleanData.map(d => {
          const isMatch = matchingSet.has(d.rank);
          const isActive = spotlight.rank === d.rank;
          return html`
            <div 
              class="platter-cover-tile ${isActive ? 'is-active-track' : ''} ${!isMatch ? 'is-dimmed' : ''}" 
              title="#${d.rank}: ${d.album} - ${d.artist} (${d.year}) | ⭐ ${d.rating.toFixed(2)}"
              onclick=${() => {
                mutable activeAlbum = d;
              }}
              onmouseover=${() => {
                mutable activeAlbum = d;
              }}
            >
              <img 
                src="${d.image_url}" 
                alt="${d.album}" 
                referrerpolicy="no-referrer"
                crossorigin="anonymous"
                onerror="this.src='https://placehold.co/100x100/181b26/00f2fe?text=Album'"
              />
            </div>
          `;
        })}
      </div>
    </div>

    <!-- Realistic Tonearm Assembly on Right Side of the Deck -->
    <div class="tonearm-column">
      <div class="tonearm-base">
        <div class="tonearm-counterweight"></div>
      </div>
      
      <div class="tonearm-rod" style="${isSpinning ? '' : 'transform: rotate(-18deg); animation: none;'}">
        <div class="tonearm-headshell">
          <div class="needle-light"></div>
        </div>
      </div>

      <div class="pitch-control">
        <div>33 ⅓ RPM</div>
        <div style="color: #00f2fe; margin-top: 4px; font-weight: bold;">● DIRECT DRIVE</div>
      </div>
    </div>
  </div>

  <!-- Bottom Console: Illuminated LCD Monitor Screen & Deck Controls -->
  <div class="turntable-lcd-screen">
    <div class="lcd-playback-line">
      <div>
        <div style="font-size: 0.7rem; color: #00f2fe; font-family: monospace; letter-spacing: 1.5px; font-weight: 700;">
          ⚡ NEEDLE DROP // ACTIVE GROOVE
        </div>
        <div class="lcd-album-title">
          ${spotlight.album}
        </div>
        <div style="font-size: 0.95rem; color: #cbd5e1; font-family: monospace;">
          ${spotlight.artist} <span style="color: #ffd166;">(${spotlight.year})</span>
        </div>
      </div>

      <!-- Equalizer Visualizer Bars -->
      <div class="equalizer-wrap" style="height: 22px; margin: 0;">
        <div class="eq-bar" style="${isSpinning ? '' : 'animation: none; height: 30%;'}"></div>
        <div class="eq-bar" style="${isSpinning ? '' : 'animation: none; height: 30%;'}"></div>
        <div class="eq-bar" style="${isSpinning ? '' : 'animation: none; height: 30%;'}"></div>
        <div class="eq-bar" style="${isSpinning ? '' : 'animation: none; height: 30%;'}"></div>
        <div class="eq-bar" style="${isSpinning ? '' : 'animation: none; height: 30%;'}"></div>
        <div class="eq-bar" style="${isSpinning ? '' : 'animation: none; height: 30%;'}"></div>
        <div class="eq-bar" style="${isSpinning ? '' : 'animation: none; height: 30%;'}"></div>
        <div class="eq-bar" style="${isSpinning ? '' : 'animation: none; height: 30%;'}"></div>
      </div>
    </div>

    <!-- Metadata Chips -->
    <div class="lcd-meta-chips">
      <span class="lcd-chip" style="color: #ffd166; border-color: rgba(255, 209, 102, 0.4);">
        ALL-TIME RANK: #${spotlight.rank} OF 1000
      </span>
      <span class="lcd-chip">
        RATING: ⭐ ${spotlight.rating ? spotlight.rating.toFixed(2) : "4.00"}
      </span>
      <span class="lcd-chip" style="text-transform: uppercase;">
        GENRE: ${spotlight.primary_genre}
      </span>
      <span class="lcd-chip" style="color: #38ef7d; border-color: rgba(56, 239, 125, 0.4);">
        GROOVES LIT: ${matchingSet.size} / 1000
      </span>
    </div>

    <!-- Turntable Action Buttons -->
    <div class="deck-controls" style="margin-top: 1rem;">
      <button 
        class="btn-neon" 
        style="background: ${isSpinning ? 'linear-gradient(135deg, #ffd166, #ff7b00)' : 'linear-gradient(135deg, #00f2fe, #4facfe)'}; font-size: 0.85rem; padding: 0.45rem 1.1rem;"
        onclick=${() => {
          mutable isSpinning = !isSpinning;
        }}
      >
        ${isSpinning ? '⏹ PAUSE PLATTER' : '▶ SPIN PLATTER'}
      </button>

      <button 
        class="btn-neon" 
        style="font-size: 0.85rem; padding: 0.45rem 1.1rem;"
        onclick=${() => {
          const randIndex = Math.floor(Math.random() * cleanData.length);
          mutable activeAlbum = cleanData[randIndex];
        }}
      >
        🎲 Drop Needle (Random)
      </button>

      <a 
        class="btn btn-outline-light btn-sm" 
        style="font-size: 0.85rem; padding: 0.45rem 1.1rem; border-radius: 8px; font-family: monospace;" 
        target="_blank" 
        href="https://open.spotify.com/search/${encodeURIComponent(spotlight.artist + ' ' + spotlight.album)}"
      >
        🎧 Listen on Spotify ↗
      </a>
    </div>
  </div>
</div>
`
