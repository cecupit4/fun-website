viewof search = Inputs.search(cleanData, {
  placeholder: "Search artist, album, genre, or release year..."
})
genresList = ["All", "rock", "pop", "folk", "electronica", "punk", "hip-hop", "soul", "jazz", "country", "metal"]
viewof genreFilter = Inputs.select(genresList, {
  label: "Genre",
  value: "All"
})
viewof limit = Inputs.select([48, 96, 240, 1000], {
  label: "Display",
  value: 48
})
// Filter data by active search & genre
filtered = search.filter(d => genreFilter === "All" || d.primary_genre === genreFilter)
