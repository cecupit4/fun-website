Top 1000 Album Covers

Explore all 1,000 greatest albums of all time unified in a modern interactive glassmorphic soundstage.

albums = FileAttachment("top_1000_albums.csv").csv({ typed: true })

cleanData = albums.map(d => ({
  ...d,
  primary_genre: d.genres ? d.genres.split(",")[0].trim() : "other",
  decade: `${Math.floor(d.year / 10) * 10}s`
}))

// State for active spotlight album
mutable activeAlbum = cleanData[0]
viewof searchQuery = Inputs.text({
  placeholder: "Search 1,000 albums by artist, title, year, or genre..."
})

genresList = ["All", "rock", "pop", "folk", "electronica", "punk", "hip-hop", "soul", "jazz", "country", "metal"]
viewof selectedGenre = Inputs.select(genresList, {
  label: "Genre Filter",
  value: "All"
})
q = searchQuery.toLowerCase().trim()
matchingSet = new Set(
  cleanData
    .filter(d => {
      const matchGenre = selectedGenre === "All" || d.primary_genre === selectedGenre;
      const matchText = !q || 
        d.album.toLowerCase().includes(q) || 
        d.artist.toLowerCase().includes(q) || 
        String(d.year).includes(q) ||
        d.primary_genre.toLowerCase().includes(q);
      return matchGenre && matchText;
    })
    .map(d => d.rank)
)

spotlight = activeAlbum || cleanData[0]
// Modern Glassmorphic Sound Prism Visual Component
html`
<div class="modern-prism-visual">
  <!-- Top Console: Integrated Vault Tabs & Modern Hero Showcase -->
  <div class="modern-showcase-header">
    <div class="deck-nav-bar" style="margin-bottom: 0.85rem; padding-bottom: 0.75rem;">
      <span style="font-size: 0.75rem; font-weight: 700; color: #ffd166; text-transform: uppercase;">VAULT TABS:</span>
      <a href="index.qmd" class="deck-nav-pill active">🌌 Sonic Matrix</a>
      <a href="compare.qmd" class="deck-nav-pill">⚖️ Cover Compare ↗</a>
      <a href="game.qmd" class="deck-nav-pill">🎮 Higher / Lower Game ↗</a>
      <a href="about.qmd" class="deck-nav-pill">ℹ️ About ↗</a>
    </div>

    <div class="modern-hero-layout">
      <!-- Modern High-Res Cover Frame with Neon Reflection -->
      <div class="modern-cover-frame">
        <img 
          src="${spotlight.image_url}" 
          alt="${spotlight.album}" 
          referrerpolicy="no-referrer"
          crossorigin="anonymous"
          onerror="this.src='https://placehold.co/300x300/181b26/00f2fe?text=Album'"
        />
      </div>

      <!-- Modern Album Metadata & Audio Spectrum -->
      <div>
        <div style="font-size: 0.72rem; letter-spacing: 2px; color: #00f2fe; text-transform: uppercase; font-weight: 800;">
          ⚡ ACTIVE ALBUM SPOTLIGHT
        </div>
        <h2 style="margin: 0.2rem 0 0.15rem 0; font-size: 1.65rem; font-weight: 800; color: #fff; line-height: 1.2;">
          ${spotlight.album}
        </h2>
        <div style="font-size: 1.05rem; color: #cbd5e1; font-weight: 500;">
          ${spotlight.artist} <span style="color: #94a3b8;">•</span> <span style="color: #ffd166;">${spotlight.year}</span>
        </div>

        <!-- Modern Stat Badges -->
        <div class="modern-stat-chips">
          <span class="modern-chip chip-accent">🏆 #${spotlight.rank} All-Time</span>
          <span class="modern-chip chip-gold">⭐ ${spotlight.rating ? spotlight.rating.toFixed(2) : "4.00"} Rating</span>
          <span class="modern-chip" style="text-transform: capitalize;">🎸 ${spotlight.primary_genre}</span>
          <span class="modern-chip" style="color: #38ef7d; border-color: rgba(56, 239, 125, 0.35); background: rgba(56, 239, 125, 0.1);">● Hi-Res Master</span>
        </div>

        <!-- Modern Cyberpunk Soundbar -->
        <div class="modern-soundbar">
          <div class="bar-cyan-magenta" style="height: 40%;"></div>
          <div class="bar-cyan-magenta" style="height: 85%;"></div>
          <div class="bar-cyan-magenta" style="height: 60%;"></div>
          <div class="bar-cyan-magenta" style="height: 100%;"></div>
          <div class="bar-cyan-magenta" style="height: 75%;"></div>
          <div class="bar-cyan-magenta" style="height: 90%;"></div>
          <div class="bar-cyan-magenta" style="height: 50%;"></div>
          <div class="bar-cyan-magenta" style="height: 95%;"></div>
          <div class="bar-cyan-magenta" style="height: 65%;"></div>
          <div class="bar-cyan-magenta" style="height: 80%;"></div>
          <div class="bar-cyan-magenta" style="height: 45%;"></div>
          <div class="bar-cyan-magenta" style="height: 70%;"></div>
        </div>

        <div class="deck-controls">
          <button class="btn-neon" style="padding: 0.4rem 1.1rem; font-size: 0.85rem;" onclick=${() => {
            const randIndex = Math.floor(Math.random() * cleanData.length);
            mutable activeAlbum = cleanData[randIndex];
          }}>
            🎲 Random Discovery
          </button>

          <a class="btn btn-outline-light btn-sm" style="font-size: 0.85rem; padding: 0.4rem 1.1rem; border-radius: 8px;" target="_blank" href="https://open.spotify.com/search/${encodeURIComponent(spotlight.artist + " " + spotlight.album)}">
            🎧 Stream on Spotify ↗
          </a>

          <span style="font-size: 0.8rem; color: #94a3b8; margin-left: auto;">
            Active Filter: <strong style="color: #00f2fe;">${matchingSet.size}</strong> of 1,000 albums
          </span>
        </div>
      </div>
    </div>
  </div>

  <!-- Modern 1,000-Cover Canvas: All covers with reliable image loading -->
  <div class="soundstage-matrix">
    ${cleanData.map(d => {
      const isMatch = matchingSet.has(d.rank);
      const isActive = spotlight.rank === d.rank;
      return html`
        <div 
          class="matrix-tile ${isActive ? 'is-active' : ''} ${!isMatch ? 'is-dimmed' : ''}" 
          title="#${d.rank}: ${d.album} by ${d.artist} (${d.year}) | ⭐ ${d.rating.toFixed(2)}"
          onclick=${() => {
            mutable activeAlbum = d;
          }}
          onmouseover=${() => {
            mutable activeAlbum = d;
          }}
        >
          <img 
            src="${d.image_url}" 
            alt="${d.album}" 
            referrerpolicy="no-referrer"
            crossorigin="anonymous"
            onerror="this.src='https://placehold.co/100x100/181b26/00f2fe?text=Album'"
          />
        </div>
      `;
    })}
  </div>
</div>
`
