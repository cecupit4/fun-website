albums = FileAttachment("top_1000_albums.csv").csv({ typed: true })

// Helper to pull the primary genre for grouping and coloring
cleanData = albums.map(d => ({
  ...d,
  primary_genre: d.genres ? d.genres.split(",")[0].trim() : "other",
  decade: `${Math.floor(d.year / 10) * 10}s`
}))

// State for currently playing album on the turntable
mutable nowPlaying = cleanData[0]
// Turntable Deck component with integrated tab navigation
html`
<div class="turntable-deck" id="turntable">
  <div class="platter-area">

      <span style="font-size: 0.75rem; font-weight: 700; color: #ffd166; text-transform: uppercase;">VAULT TABS:</span>
      <a href="index.qmd" class="deck-nav-pill active">🎵 Turntable Wall</a>
      <a href="compare.qmd" class="deck-nav-pill">⚖️ Cover Compare ↗</a>

viewof search = Inputs.search(cleanData, {
  placeholder: "Search artist, album, genre, or release year..."
})
genresList = ["All", "rock", "pop", "folk", "electronica", "punk", "hip-hop", "soul", "jazz", "country", "metal"]
viewof genreFilter = Inputs.select(genresList, {
  label: "Genre",
  value: "All"
})
viewof mosaicLimit = Inputs.select([72, 144, 300, 1000], {
  label: "Covers",
  value: 144
})
// Filter data by active search & genre
filtered = search.filter(d => genreFilter === "All" || d.primary_genre === genreFilter)
md`### Visual Cover Mosaic (*${Math.min(mosaicLimit, filtered.length)} of ${filtered.length} matching covers — click any to play on turntable*)`
// Pure visual mosaic of album covers without any bottom card list
html`<div class="mosaic-wall">
  ${filtered.slice(0, mosaicLimit).map(d => html`
    <div class="mosaic-tile" title="#${d.rank}: ${d.album} - ${d.artist} (${d.year}) | ⭐ ${d.rating.toFixed(2)}" onclick=${() => {
      mutable nowPlaying = d;
      document.getElementById('turntable').scrollIntoView({ behavior: 'smooth' });
    }}>
      <img src="${d.image_url}" alt="${d.album}" loading="lazy" />
      <span class="mosaic-rank">#${d.rank}</span>
    </div>
  `)}
</div>`
