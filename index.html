Top 1000 Album Covers

Explore all 1,000 greatest albums of all time unified in a single interactive visual soundstage.

albums = FileAttachment("top_1000_albums.csv").csv({ typed: true })

cleanData = albums.map(d => ({
  ...d,
  primary_genre: d.genres ? d.genres.split(",")[0].trim() : "other",
  decade: `${Math.floor(d.year / 10) * 10}s`
}))

// State for currently playing album on the turntable
mutable nowPlaying = cleanData[0]
viewof searchQuery = Inputs.text({
  placeholder: "Search 1,000 albums by artist, title, year, or genre..."
})

genresList = ["All", "rock", "pop", "folk", "electronica", "punk", "hip-hop", "soul", "jazz", "country", "metal"]
viewof selectedGenre = Inputs.select(genresList, {
  label: "Genre Filter",
  value: "All"
})
q = searchQuery.toLowerCase().trim()
matchingSet = new Set(
  cleanData
    .filter(d => {
      const matchGenre = selectedGenre === "All" || d.primary_genre === selectedGenre;
      const matchText = !q || 
        d.album.toLowerCase().includes(q) || 
        d.artist.toLowerCase().includes(q) || 
        String(d.year).includes(q) ||
        d.primary_genre.toLowerCase().includes(q);
      return matchGenre && matchText;
    })
    .map(d => d.rank)
)
// Unified Soundstage Visual that integrates the turntable and all 1,000 album covers into one cohesive graphic
html`
<div class="soundstage-visual">
  <!-- Top Console: Integrated Vault Tabs & Active Turntable Deck -->
  <div class="soundstage-header">
    <div class="deck-nav-bar" style="margin-bottom: 0.85rem; padding-bottom: 0.75rem;">
      <span style="font-size: 0.75rem; font-weight: 700; color: #ffd166; text-transform: uppercase;">VAULT TABS:</span>
      <a href="index.qmd" class="deck-nav-pill active">🎵 1000-Cover Galaxy</a>
      <a href="compare.qmd" class="deck-nav-pill">⚖️ Cover Compare ↗</a>
      <a href="game.qmd" class="deck-nav-pill">🎮 Higher / Lower Game ↗</a>
      <a href="about.qmd" class="deck-nav-pill">ℹ️ About ↗</a>
    </div>

    <div class="soundstage-deck-layout">
      <!-- Mini Spinning Turntable Platter -->
      <div style="text-align: center;">
        <div class="vinyl-record" style="width: 120px; height: 120px; margin: 0 auto; box-shadow: 0 0 20px rgba(0, 242, 254, 0.25);">
          <div class="vinyl-label" style="width: 50px; height: 50px;">
            <img src="${nowPlaying.image_url}" alt="${nowPlaying.album}" />
          </div>
        </div>
      </div>

      <!-- Active Album Details & Equalizer -->
      <div>
        <div style="font-size: 0.72rem; letter-spacing: 2px; color: #00f2fe; text-transform: uppercase; font-weight: 700;">
          ⚡ Active Turntable Spindle
        </div>
        <h3 style="margin: 0.2rem 0 0.1rem 0; font-size: 1.45rem; font-weight: 800; color: #fff; line-height: 1.2;">
          ${nowPlaying.album}
        </h3>
        <div style="font-size: 0.95rem; color: #cbd5e1;">
          ${nowPlaying.artist} <span style="color: #ffd166;">(${nowPlaying.year})</span>
        </div>
        <div style="margin-top: 0.35rem; font-size: 0.82rem; color: #94a3b8;">
          All-Time Rank: <strong style="color: #00f2fe;">#${nowPlaying.rank}</strong> of 1,000 &nbsp;|&nbsp;
          Rating: <strong style="color: #ffd166;">⭐ ${nowPlaying.rating.toFixed(2)}</strong> &nbsp;|&nbsp;
          Genre: <strong style="color: #e2e8f0; text-transform: capitalize;">${nowPlaying.primary_genre}</strong>
        </div>

        <div class="equalizer-wrap" style="height: 18px; margin: 0.5rem 0;">
          <div class="eq-bar"></div>
          <div class="eq-bar"></div>
          <div class="eq-bar"></div>
          <div class="eq-bar"></div>
          <div class="eq-bar"></div>
          <div class="eq-bar"></div>
          <div class="eq-bar"></div>
          <div class="eq-bar"></div>
        </div>

        <div class="deck-controls" style="margin-top: 0.6rem;">
          <button class="btn-neon" style="padding: 0.35rem 0.9rem; font-size: 0.8rem;" onclick=${() => {
            const randIndex = Math.floor(Math.random() * cleanData.length);
            mutable nowPlaying = cleanData[randIndex];
          }}>
            🎲 Drop Needle (Random)
          </button>

          <a class="btn btn-outline-light btn-sm" style="font-size: 0.8rem; padding: 0.35rem 0.9rem; border-radius: 8px;" target="_blank" href="https://open.spotify.com/search/${encodeURIComponent(nowPlaying.artist + " " + nowPlaying.album)}">
            🎧 Listen on Spotify ↗
          </a>

          <span style="font-size: 0.8rem; color: #94a3b8; margin-left: auto; align-self: center;">
            🌌 Glowing: <strong style="color: #00f2fe;">${matchingSet.size}</strong> of 1,000 covers
          </span>
        </div>
      </div>
    </div>
  </div>

  <!-- The Visual Canvas: All 1,000 Album Covers integrated directly inside the visual -->
  <div class="soundstage-matrix">
    ${cleanData.map(d => {
      const isMatch = matchingSet.has(d.rank);
      const isActive = nowPlaying.rank === d.rank;
      return html`
        <div 
          class="matrix-tile ${isActive ? 'is-active' : ''} ${!isMatch ? 'is-dimmed' : ''}" 
          title="#${d.rank}: ${d.album} by ${d.artist} (${d.year}) | ⭐ ${d.rating.toFixed(2)}"
          onclick=${() => {
            mutable nowPlaying = d;
          }}
          onmouseover=${() => {
            mutable nowPlaying = d;
          }}
        >
          <img src="${d.image_url}" alt="${d.album}" loading="lazy" />
        </div>
      `;
    })}
  </div>
</div>
`
